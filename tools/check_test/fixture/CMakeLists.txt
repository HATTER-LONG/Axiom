cmake_minimum_required(VERSION 3.25)
project(check_test_fixture LANGUAGES CXX)

set(FIXTURE_SANITIZERS "" CACHE STRING "Comma-separated sanitizer list")
option(FIXTURE_COVERAGE "Enable LLVM source coverage" OFF)

add_library(answer library/answer.cpp)
target_include_directories(answer PUBLIC library)
target_compile_features(answer PUBLIC cxx_std_20)

include(CTest)
if(BUILD_TESTING)
  add_executable(answer_test spec/answer_test.cpp)
  target_link_libraries(answer_test PRIVATE answer)
  add_test(NAME answer COMMAND answer_test)
endif()

set(fixture_targets answer)
if(TARGET answer_test)
  list(APPEND fixture_targets answer_test)
endif()

foreach(fixture_target IN LISTS fixture_targets)
  if(FIXTURE_COVERAGE)
    target_compile_options(${fixture_target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(${fixture_target} PRIVATE -fprofile-instr-generate)
  endif()
  if(FIXTURE_SANITIZERS)
    target_compile_options(${fixture_target} PRIVATE -fsanitize=${FIXTURE_SANITIZERS} -fno-omit-frame-pointer)
    target_link_options(${fixture_target} PRIVATE -fsanitize=${FIXTURE_SANITIZERS})
  endif()
endforeach()

if(WIN32 AND FIXTURE_SANITIZERS MATCHES "address" AND TARGET answer_test)
  get_filename_component(fixture_llvm_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  get_filename_component(fixture_llvm_root "${fixture_llvm_bin}" DIRECTORY)
  file(GLOB fixture_runtime_dirs LIST_DIRECTORIES true "${fixture_llvm_root}/lib/clang/*/lib/windows")
  find_file(fixture_asan_runtime NAMES clang_rt.asan_dynamic-x86_64.dll PATHS ${fixture_runtime_dirs} NO_DEFAULT_PATH)
  if(NOT fixture_asan_runtime)
    message(FATAL_ERROR "Clang ASan runtime DLL was not found")
  endif()
  add_custom_command(TARGET answer_test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${fixture_asan_runtime}" "$<TARGET_FILE_DIR:answer_test>")
endif()
